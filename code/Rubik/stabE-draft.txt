             else all (fun u0 => let ui := of_nat u0 in
                    if (ui =? fsymi r0)%uint63 then true
                    else if (ract r0 ui =? repsi (frepi r0))%uint63 then
                      all_pow ntwbits 0%uint63
                        (fun tw0 => if (ntwisti <=? tw0)%uint63 then true

                    else true)
                  (iota 0 16)).
